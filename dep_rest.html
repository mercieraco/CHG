`).bind(doctorId).all(); // สร้าง array ของวันที่ทั้งหมด พร้อม duties const dates = []; const start = new Date(startDate); const end = new Date(endDate); for (let d = new Date(start); d <= end; d.setDate(d.getDate() + 1)) { const dateStr = d.toISOString().split('T')[0]; const dayOfWeek = d.getDay(); // 0=อาทิตย์, 1=จันทร์, ..., 6=เสาร์ // หา duties สำหรับวันนี้ const dutiesForDay = []; dutyTemplates.forEach(template => { let hasDuty = false; switch(dayOfWeek) { case 0: hasDuty = template.sunday === 1; break; case 1: hasDuty = template.monday === 1; break; case 2: hasDuty = template.tuesday === 1; break; case 3: hasDuty = template.wednesday === 1; break; case 4: hasDuty = template.thursday === 1; break; case 5: hasDuty = template.friday === 1; break; case 6: hasDuty = template.saturday === 1; break; } if (hasDuty) { dutiesForDay.push({ duty_id: template.duty_id, duty_name: template.duty_name }); } }); dates.push({ duty_date: dateStr, duties: dutiesForDay }); } return new Response(JSON.stringify({ success: true, data: dates }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } // ===================================== // 1.5. GET /api/leaves/department/:id (แก้ไข) // ดึงการลาพร้อม duty details // ===================================== if (url.pathname.match(/^\/api\/leaves\/department\/\d+$/) && request.method === 'GET') { const departmentId = url.pathname.split('/')[4]; // ดึงการลาทั้งหมด const { results } = await env.DB.prepare(` SELECT l.*, d.name as doctor_name, d.department_id, rd.name as replacement_doctor_name FROM doctor_leaves l JOIN doctors d ON l.doctor_id = d.id LEFT JOIN doctors rd ON l.replacement_doctor_id = rd.id WHERE d.department_id = ? ORDER BY l.start_date DESC `).bind(departmentId).all(); // สำหรับแต่ละการลา ดึง duty details (ถ้าเป็น multiple) for (const leave of results) { if (leave.delegation_type === 'multiple') { const { results: duties } = await env.DB.prepare(` SELECT ld.*, duty.duty_name, rd.name as replacement_doctor_name FROM leave_duties ld JOIN duties duty ON ld.duty_id = duty.id LEFT JOIN doctors rd ON ld.replacement_doctor_id = rd.id WHERE ld.leave_id = ? ORDER BY ld.duty_date, duty.duty_name `).bind(leave.id).all(); leave.duty_assignments = duties; } } return new Response(JSON.stringify({ success: true, data: results }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } // ===================================== // 3.5. GET /api/leave-duties/department/:departmentId // ดึง leave_duties ทั้งหมดในแผนก (ใช้เช็คแพทย์ที่ได้รับมอบหมาย) // ===================================== if (url.pathname.match(/^\/api\/leave-duties\/department\/\d+$/) && request.method === 'GET') { const departmentId = url.pathname.split('/')[4]; const { results } = await env.DB.prepare(` SELECT ld.leave_id, ld.duty_id, ld.duty_date, ld.replacement_doctor_id, l.doctor_id as leave_doctor_id, l.start_date, l.end_date, duty.duty_name, rd.name as replacement_doctor_name FROM leave_duties ld JOIN doctor_leaves l ON ld.leave_id = l.id JOIN doctors d ON l.doctor_id = d.id JOIN duties duty ON ld.duty_id = duty.id LEFT JOIN doctors rd ON ld.replacement_doctor_id = rd.id WHERE d.department_id = ? `).bind(departmentId).all(); return new Response(JSON.stringify({ success: true, data: results }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } // ===================================== // 4. DELETE /api/leaves/:id // ===================================== if (url.pathname.match(/^\/api\/leaves\/\d+$/) && request.method === 'DELETE') { const leaveId = url.pathname.split('/')[3]; // ลบ leave_duties (cascade จะลบอัตโนมัติ แต่ manual ก็ได้) await env.DB.prepare('DELETE FROM leave_duties WHERE leave_id = ?') .bind(leaveId).run(); // ลบ doctor_leaves await env.DB.prepare('DELETE FROM doctor_leaves WHERE id = ?') .bind(leaveId).run(); return new Response(JSON.stringify({ success: true }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } // Route not found return new Response(JSON.stringify({ success: false, message: 'Route not found' }), { status: 404, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } catch (error) { return new Response(JSON.stringify({ success: false, message: error.message }), { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } ``` } };
